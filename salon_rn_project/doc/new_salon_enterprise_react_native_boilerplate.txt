// tokenService.ts
import * as Keychain from 'react-native-keychain';

export const setTokens = async (access, refresh) => {
  await Keychain.setGenericPassword('auth', JSON.stringify({ access, refresh }));
};

export const getTokens = async () => {
  const data = await Keychain.getGenericPassword();
  return data ? JSON.parse(data.password) : null;
};

🧠 RBAC PERMISSION MATRIX

export const PERMISSIONS = {
  SUPER_ADMIN: ['*'],
  OWNER: ['manage_salon','view_reports','manage_staff'],
  STAFF: ['view_bookings','update_booking'],
  CUSTOMER: ['create_booking'],
  VENDOR: ['manage_inventory']
};

export const hasPermission = (role, permission) => {
  return PERMISSIONS[role]?.includes('*') || PERMISSIONS[role]?.includes(permission);
};

// RootNavigator.tsx
import { useAuthStore } from '@/store/authStore';

export default function RootNavigator() {
  const { role } = useAuthStore();

  if (!role) return <AuthStack />;

  const map = {
    SUPER_ADMIN: <AdminStack />,
    OWNER: <OwnerStack />,
    STAFF: <StaffStack />,
    CUSTOMER: <CustomerStack />,
    VENDOR: <OwnerStack />
  };

  return map[role];
}
